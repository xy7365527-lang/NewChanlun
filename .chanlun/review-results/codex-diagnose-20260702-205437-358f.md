## C. MuEstimator（mu_estimator.rs）——merge / observe / trades 关键片段

```rust
#[derive(Debug, Clone, Default)]
pub struct MuEstimator {
    // ponytail: 全量逐笔留存供 perm_test 置换（Welford 聚合量算不出置换）；OOS BTC 数万笔可接受。
    trades: Vec<(MuClass, f64)>,
    buckets: HashMap<MuClass, Welford>,
}

impl MuEstimator {
    pub fn observe(&mut self, obs: MuObservation) {
        self.buckets.entry(obs.class).or_default().push(obs.x_gamma);
        self.trades.push((obs.class, obs.x_gamma));
    }

    pub fn trades(&self) -> &[(MuClass, f64)] { &self.trades }

    /// 合并另一估计器的全部桶（Chan/Welford 并行合并，bit-exact 等价逐笔顺序累加同一桶）。
    /// 用途（cross-fit OOS，acc-crossfit-oos）：K-fold 的 train 是**不连续时间块**——各连续段独立
    /// 估 μ 后合并到一个 est（**不拼接成单 Dataset**，防接缝伪相邻）。
    /// 合并公式（Chan et al. 1979 并行 Welford）：
    /// n=n_a+n_b, δ=mean_b−mean_a, mean=mean_a+δ·n_b/n, m2=m2_a+m2_b+δ²·n_a·n_b/n。
    pub fn merge(&mut self, other: &MuEstimator) {
        for (z, wb) in &other.buckets {
            let wa = self.buckets.entry(*z).or_default();
            if wb.n == 0 { continue; }
            if wa.n == 0 { *wa = *wb; continue; }
            let n = wa.n + wb.n;
            let delta = wb.mean - wa.mean;
            let mean = wa.mean + delta * (wb.n as f64) / (n as f64);
            let m2 = wa.m2 + wb.m2 + delta * delta * (wa.n as f64) * (wb.n as f64) / (n as f64);
            *wa = Welford { n, mean, m2 };
        }
        // 注意：merge 不 touch self.trades——other.trades 完全被丢弃，这正是 wverify_full 首跑
        // panic「trades 恒空」的根源（原代码在 walk_forward_oos_mu 里对每窗 est_i 调 agg.merge(&est_i)）。
    }
}
```

## D. decontam.rs——effective_n / powered / classify_bucket（完整）

```rust
/// n_eff = n / τ，τ = 1 + 2·Σ_{k≥1}ρk（积分自相关时间，Geyer 配对 IPS 截断）。
/// 序列须按**成交时间序**传入（模块文档原话："wverify_run 逐 bar 因果收集"）。
pub fn effective_n(x: &[f64]) -> f64 {
    let n = x.len();
    if n < 2 { return n as f64; }
    if x.iter().any(|v| !v.is_finite()) { return f64::NAN; }
    let mean = x.iter().sum::<f64>() / n as f64;
    let c0: f64 = x.iter().map(|v| (v - mean).powi(2)).sum();
    if c0 == 0.0 { return n as f64; }
    let max_lag = n / 2;
    let rho = |k: usize| -> f64 {
        (0..n - k).map(|i| (x[i] - mean) * (x[i + k] - mean)).sum::<f64>() / c0
    };
    let sum_rho = geyer_paired_sum(rho, max_lag);
    let tau = (1.0 + 2.0 * sum_rho).max(1.0);
    n as f64 / tau
}

pub fn powered(n_eff: f64, cv: f64, z_alpha: f64) -> bool {
    n_eff >= (z_alpha * cv).powi(2)
}

pub fn classify_bucket(mu_hat: f64, lcb: f64, ucb: f64, perm_p: f64, n_eff: f64, cv: f64, z_alpha: f64, perm_alpha: f64) -> AlphaState {
    let powered = powered(n_eff, cv, z_alpha);
    if powered && mu_hat > 0.0 && perm_p < perm_alpha && lcb > 0.0 { AlphaState::Validated }
    else if powered && lcb <= 0.0 && ucb <= 0.0 { AlphaState::Falsified }
    else { AlphaState::Inconclusive }
}

pub fn global_verdict(states: &[AlphaState]) -> AcceptanceVerdict {
    if states.iter().any(|s| *s == AlphaState::Validated) { AcceptanceVerdict::Pass }
    else if states.is_empty() || states.iter().any(|s| *s == AlphaState::Inconclusive) { AcceptanceVerdict::Inconclusive }
    else { AcceptanceVerdict::Falsified }
}
```
