## `extract_c_segment_timestamps` 实际代码

```python
def extract_c_segment_timestamps(
    move: Move,
    segments: list,
    bars: list[Bar],
) -> tuple[datetime, datetime]:
    # 边界检查
    if not segments: raise ValueError("segments must not be empty")
    if not bars: raise ValueError("bars must not be empty")
    if move.seg_start < 0 or move.seg_end < move.seg_start:
        raise ValueError(f"invalid seg range: ...")
    if move.seg_end >= len(segments):
        raise IndexError(f"seg_end={move.seg_end} exceeds segments length=...")

    # 本地重建 merged_to_raw（幂等）
    import pandas as pd
    from newchan.a_inclusion import merge_inclusion
    df_raw = pd.DataFrame({...}, index=[b.ts for b in bars])
    _, merged_to_raw = merge_inclusion(df_raw)

    if not merged_to_raw: raise ValueError("merge_inclusion returned empty merged_to_raw")

    seg_first = segments[move.seg_start]
    seg_last = segments[move.seg_end]
    n_merged = len(merged_to_raw)
    n_raw = len(bars)

    i0 = max(0, min(seg_first.i0, n_merged - 1))
    i1 = max(0, min(seg_last.i1, n_merged - 1))

    raw_start = merged_to_raw[i0][0]   # 起点段：合并块起始 raw
    raw_end = merged_to_raw[i1][1]     # 终点段：合并块终点 raw
    raw_start = max(0, min(raw_start, n_raw - 1))
    raw_end = max(0, min(raw_end, n_raw - 1))

    return bars[raw_start].ts, bars[raw_end].ts
```
