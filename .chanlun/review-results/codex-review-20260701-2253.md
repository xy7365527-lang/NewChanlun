### C. build_nest_certificate 实现（econ_positive.rs:481-547）

```rust
pub(super) fn build_nest_certificate(
    tower: &[Rc<Vec<LeveledMove>>],
    lvl: usize,
    source_index: usize,
    delta: Side,
    bits: &BspBits,
    hist: &[f64],
) -> Option<NestCertificate> {
    // 执行级 tower[lvl] 中找 end_index == source_index 的段
    let exec_moves = tower.get(lvl)?.as_slice();
    let s = exec_moves.iter().find(|m| m.end_index == source_index)?;
    let base_interval = NestInterval {
        start_time: s.start_index as u64,
        end_time: s.end_index as u64,
        idx: s.id.ordinal,
    };

    // rungs：从高级向执行级降序
    let max_k = tower.len();
    let mut rung_buf: Vec<NestRung> = Vec::new();
    for k in (lvl + 1)..max_k {   // ← 关键：遍历所有上级
        let k_moves = tower[k].as_slice();
        let Some(knode) = k_moves.iter().find(|m| {
            m.start_index <= source_index && source_index <= m.end_index
        }) else {
            break; // 无 k 级包含段 ⟹ 链断
        };
        let interval_k = NestInterval { ... };
        // Cand^δ_k：计算 DivCand 四条件
        let ctx: Vec<ContextMove> = knode.sub_moves.iter().map(|m| ...).collect();
        let target_idx = knode.sub_moves.iter().position(|m| m.end_index == source_index);
        let cand_k = match target_idx {
            Some(tidx) => div_cand(&DivCandInput { context: &ctx, target_idx: tidx, hist, delta }),
            None => false,
        };
        rung_buf.push(NestRung { interval: interval_k, cand: cand_k });
    }
    rung_buf.reverse();
    Some(NestCertificate {
        side: delta,
        terminal: *bits,
        base_interval,
        rungs: rung_buf,   // ← 多级 rungs
    })
}
```

### D. n_delta 递归实现（nest.rs:200-218）

```rust
pub fn n_delta(&self) -> bool {
    Self::n_delta_rec(self.side, &self.terminal, &self.base_interval, &self.rungs)
}

fn n_delta_rec(side: Side, terminal: &BspBits, base: &NestInterval, rungs: &[NestRung]) -> bool {
    match rungs.split_first() {
        // 基例 ℓ=e：N^δ_{e↓e} = Conf^δ_e
        None => terminal.confirm_side(side),
        // 递归步 ℓ>e：Cand^δ_ℓ ∧ [J^δ_{ℓ-1} ⊆ J^δ_ℓ] ∧ N^δ_{ℓ-1↓e}
        Some((top, rest)) => {
            let child = rest.first().map_or(base, |r| &r.interval);
            top.cand
                && is_sub(child, &top.interval)  // ← J嵌套检查
                && Self::n_delta_rec(side, terminal, base, rest)  // ← 真递归
        }
    }
}
```

### E. 单级退化情形分析

当 `tower.len() == lvl + 1`（单层塔）：
- `for k in (lvl+1)..max_k` 循环不执行
- `rung_buf = []`（空 rungs）
- `NestCertificate { rungs: vec![], ... }` → `n_delta_rec` → `None` → `terminal.confirm_side(side)` = 纯基例

这是"有效退化"（base-case），而不是"零调用"——函数确实被调用，只是 rungs 为空时退化到单级 Conf^δ。

### F. tower 的实际层数来源

`tower_i` = `classify_at(i)` 返回的 `Vec<Rc<Vec<LeveledMove>>>`。
`classify_at` 签名：
```rust
pub fn classify_at(&mut self, i: usize) -> (classifier::Classification, Vec<Rc<Vec<LeveledMove>>>)
```
tower 的长度 = 分类后产生的级别数。当数据足够，tower 可有 2+ 层，此时 rung 循环真执行多级。
