    let threshold = (z_alpha * cv).powi(2);
    (n_eff as f64) >= threshold
}
pub fn classify_bucket(mu_hat: f64, lcb: f64, ucb: f64, perm_p: f64, n_eff: usize, cv: f64, z_alpha: f64, perm_alpha: f64) -> AlphaState {

    if powered && mu_hat > 0.0 && perm_p < perm_alpha && lcb > 0.0 { AlphaState::Validated }
    else if powered && lcb <= 0.0 && ucb <= 0.0 { AlphaState::Falsified }

}

## 5. 多重检验问题

- 判据本身对每桶独立跑 `perm_p<0.05`，未做跨桶多重检验校正（Bonferroni/FDR/family-wise）。
- 但也要考虑：`(ℓ,δ,bsp_class)` 桶键是缠论先验给定的分类格（非从收益里挑出来的），预注册 §1.1 用"先验桶键非从收益挑选"来论证这不引入选择偏差（665 命题①）。问题是"先验桶键排除选择偏差"是否等价于"5 桶同测不需要多重检验校正"——这是两个不同的统计问题（前者是"桶怎么定义"，后者是"多少次独立测试同时做显著性判断"）。
