def strip(v,ids):
    if isinstance(v,dict): return {k:strip(x,ids) for k,x in v.items() if k not in DROP}
    if isinstance(v,list): return [strip(x,ids) for x in v]
    if isinstance(v,str) and v in ids: return ids[v]
    return v
def norm(s):
    ids={}
    for key in ("objects","withdrawn_objects"):
        for o in s.get(key) or []:
            if "object_id" in o:
                ids[o["object_id"]]="ID:"+json.dumps(strip({k:v for k,v in o.items() if k!="object_id"},{}),sort_keys=True)[:400]
    out={}
    for key in ("objects","withdrawn_objects","relations","witnesses"):
        out[key]=collections.Counter(json.dumps(strip(o,ids),sort_keys=True) for o in s.get(key) or [])
    return out
