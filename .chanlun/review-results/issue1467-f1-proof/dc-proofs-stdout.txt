def DCSpec.DC_CANON_TARGET : Prop :=
∀ (p : Nat → Int) (delta : Int),
  0 < delta →
    ∀ (len : Nat),
      ∃ result,
        DCSpec.Whole p delta len result ∧ ∀ (other : DCSpec.Result), DCSpec.Whole p delta len other → other = result
def DCSpec.DC_CAUSAL_TARGET : Prop :=
∀ (p q : Nat → Int) (delta : Int),
  0 < delta →
    ∀ (len₁ len₂ t : Nat) (r₁ r₂ : DCSpec.Result),
      t < len₁ →
        t < len₂ →
          (∀ (i : Nat), i ≤ t → p i = q i) →
            DCSpec.Whole p delta len₁ r₁ →
              DCSpec.Whole q delta len₂ r₂ →
                List.filter (fun u => decide (u.knownAt ≤ t)) (DCSpec.units r₁) =
                  List.filter (fun u => decide (u.knownAt ≤ t)) (DCSpec.units r₂)
def DCSpec.DC_GEOMETRY_TARGET : Prop :=
∀ (p : Nat → Int) (delta : Int),
  0 < delta →
    ∀ (len : Nat) (result : DCSpec.Result),
      DCSpec.Whole p delta len result →
        (∀ (u : DCSpec.Unit), u ∈ DCSpec.units result → DCSpec.UnitGeometry p delta len u) ∧
          List.Pairwise (fun a b => a.knownAt < b.knownAt) (DCSpec.units result) ∧
            ∀ (pre : List DCSpec.Unit) (a b : DCSpec.Unit) (post : List DCSpec.Unit),
              DCSpec.units result = pre ++ a :: b :: post → a.finish = b.start ∧ DCSpec.flip a.direction = b.direction
